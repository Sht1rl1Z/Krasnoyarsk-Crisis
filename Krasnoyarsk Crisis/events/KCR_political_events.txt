country_event = {
    id = kcr.7

    title = kcr.7.t
    desc = kcr.7.d

    picture = GFX_KCR_event_7

    is_triggered_only = yes
    # ПОБЕДА ГУБЕРНАТОРА
    option = {
        name = kcr.7.a
        
        set_country_flag = KCR_governor_won
        set_country_flag = KCR_cabinet_conflict_finished

        set_variable = {
            KCR_cabinet_tension = 0
        }

        remove_ideas = KCR_municipal_power
        add_ideas = KCR_regional_vertical

        if = {
            limit = {
                has_country_flag = KCR_party_governor_won
            }
            complete_national_focus = KCR_gubernator_demokrat
        }
        else_if = {
            limit = {
                has_country_flag = KCR_party_nationalists_won
            }
            complete_national_focus = KCR_nacionalizm_nashe_vse
        }
        else_if = {
            limit = {
                has_country_flag = KCR_party_justice_won
            }
            complete_national_focus = KCR_spravedlivye_resheniya
        }
        else_if = {
            limit = {
                has_country_flag = KCR_party_proletariat_won
            }
            complete_national_focus = KCR_vlast_proletariatu
        }
    }
}
